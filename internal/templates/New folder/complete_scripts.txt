// SEARCH FUNCTIONS
// ============================================
let searchTimeout = null;
let selectedSearchIndex = -1;
let currentSearchResults = [];

function handleSearchInput(query) {
    clearTimeout(searchTimeout);
    const trimmed = query.trim();
    if (trimmed.length < 2) {
        closeSearchDropdown();
        return;
    }
    searchTimeout = setTimeout(function() {
        fetchSearchResults(trimmed);
    }, 200);
}

async function fetchSearchResults(query) {
    try {
        const res = await fetch('/api/products/search?q=' + encodeURIComponent(query));
        if (!res.ok) return;
        currentSearchResults = await res.json();
        renderSearchDropdown(currentSearchResults);
    } catch (err) {
        console.error("Search error:", err);
    }
}

function renderSearchDropdown(items) {
    const dropdown = document.getElementById('search-results-dropdown');
    selectedSearchIndex = -1;
    if (!items || items.length === 0) {
        dropdown.innerHTML = '<div class="p-3 text-center text-sm text-gray-500">No products found</div>';
        dropdown.classList.remove('hidden');
        return;
    }
    let html = '';
    items.forEach((item, index) => {
        const price = item.retail_price || 0;
        html += '<div id="search-item-' + index + '" onclick="selectSearchResult(' + index + ')" class="p-3 hover:bg-purple-50 cursor-pointer flex justify-between items-center transition">' +
            '<div><div class="font-bold text-gray-900 text-sm">' + item.name + '</div><div class="text-xs text-gray-500">Barcode: ' + (item.barcode || 'N/A') + ' | Stock: ' + item.stock_quantity + '</div></div>' +
            '<div class="text-right"><div class="font-bold text-purple-700 text-sm">KES ' + price.toFixed(2) + '</div><span class="text-[10px] bg-purple-100 text-purple-800 font-bold px-2 py-0.5 rounded-full">+ Add</span></div>' +
        '</div>';
    });
    dropdown.innerHTML = html;
    dropdown.classList.remove('hidden');
}

function handleSearchKeyDown(e) {
    const dropdown = document.getElementById('search-results-dropdown');
    const isDropdownVisible = !dropdown.classList.contains('hidden');
    if (!isDropdownVisible || currentSearchResults.length === 0) return;
    if (e.key === 'ArrowDown') {
        e.preventDefault();
        if (selectedSearchIndex < currentSearchResults.length - 1) {
            selectedSearchIndex++;
            updateSearchHighlight();
        }
    } else if (e.key === 'ArrowUp') {
        e.preventDefault();
        if (selectedSearchIndex > 0) {
            selectedSearchIndex--;
            updateSearchHighlight();
        }
    } else if (e.key === 'Enter') {
        if (selectedSearchIndex >= 0 && selectedSearchIndex < currentSearchResults.length) {
            e.preventDefault();
            selectSearchResult(selectedSearchIndex);
        }
    } else if (e.key === 'Escape') {
        closeSearchDropdown();
    }
}

function updateSearchHighlight() {
    currentSearchResults.forEach(function(_, index) {
        const itemEl = document.getElementById('search-item-' + index);
        if (itemEl) {
            if (index === selectedSearchIndex) {
                itemEl.classList.add('bg-purple-100');
            } else {
                itemEl.classList.remove('bg-purple-100');
            }
        }
    });
}

function selectSearchResult(index) {
    const product = currentSearchResults[index];
    if (!product) return;
    const barcodeInput = document.getElementById('barcode-input');
    if (barcodeInput && product.barcode) {
        barcodeInput.value = product.barcode;
        handleAddToCart();
    }
    const input = document.getElementById('scan-input');
    if (input) {
        input.value = '';
        input.focus();
    }
    closeSearchDropdown();
}

function closeSearchDropdown() {
    const dropdown = document.getElementById('search-results-dropdown');
    if (dropdown) {
        dropdown.classList.add('hidden');
        dropdown.innerHTML = '';
    }
    selectedSearchIndex = -1;
    currentSearchResults = [];
}

// ADD TO CART
// ============================================
let totalAmount = 0;

async function handleAddToCart() {
    const barcode = document.getElementById('barcode-input').value.trim();
    const qty = parseInt(document.getElementById('qty-input').value) || 1;
    if (!barcode) {
        alert('Please enter a barcode or search for a product.');
        return;
    }
    const existingRow = document.querySelector(`#cart-items tr[data-barcode="${barcode}"]`);
    if (existingRow) {
        const qtyInput = existingRow.querySelector('.cart-qty-input');
        const currentQty = parseInt(qtyInput.value) || 1;
        qtyInput.value = currentQty + qty;
        qtyInput.dispatchEvent(new Event('input', { bubbles: true }));
        document.getElementById('barcode-input').value = '';
        document.getElementById('qty-input').value = 1;
        document.getElementById('barcode-input').focus();
        return;
    }
    await fetchProductByBarcode(barcode, qty);
}

async function fetchProductByBarcode(barcode, qty) {
    try {
        const url = '/api/products/scan-html?barcode=' + encodeURIComponent(barcode) + '&qty=' + qty;
        const response = await fetch(url);
        if (!response.ok) {
            alert('Product not found!');
            document.getElementById('barcode-input').focus();
            return;
        }
        const html = await response.text();
        const trMatch = html.match(/<tr[\s\S]*?<\/tr>/);
        if (!trMatch) {
            alert('Error adding product to cart. Please try again.');
            return;
        }
        const trHTML = trMatch[0];
        const temp = document.createElement('tbody');
        temp.innerHTML = trHTML;
        const newRow = temp.firstElementChild;
        if (newRow && newRow.tagName === 'TR') {
            newRow.setAttribute('data-barcode', barcode);
            document.getElementById('cart-items').appendChild(newRow);
            updateCartTotals();
        } else {
            alert('Error adding product to cart. Please try again.');
        }
        document.getElementById('barcode-input').value = '';
        document.getElementById('qty-input').value = 1;
        document.getElementById('barcode-input').focus();
    } catch (error) {
        console.error('Error adding product:', error);
        alert('Error adding product: ' + error.message);
    }
}

document.getElementById('barcode-input').addEventListener('keydown', function(e) {
    if (e.key === 'Enter') {
        e.preventDefault();
        handleAddToCart();
    }
});

document.getElementById('qty-input').addEventListener('keydown', function(e) {
    if (e.key === 'Enter') {
        e.preventDefault();
        handleAddToCart();
    }
});

function updateCartTotals() {
    let total = 0;
    const rows = document.querySelectorAll('#cart-items tr');
    rows.forEach(row => {
        const subtotalTd = row.querySelector('.subtotal-td');
        if (subtotalTd) {
            const priceText = subtotalTd.innerText.replace('KES', '').trim();
            const price = parseFloat(priceText);
            if (!isNaN(price)) total += price;
        }
    });
    totalAmount = total;
    document.getElementById('grand-total').innerText = 'KES ' + total.toFixed(2);
}

function updateRowTotals(inputElement, isManualPriceChange) {
    const row = inputElement.closest('tr');
    const qtyInput = row.querySelector('.cart-qty-input');
    const priceInput = row.querySelector('.cart-price-input');
    let qty = parseInt(qtyInput.value);
    if (isNaN(qty) || qty < 1) qty = 1;
    let currentPrice = parseFloat(priceInput.value);
    if (isNaN(currentPrice) || currentPrice < 0) currentPrice = 0;
    const wholesaleQty = parseInt(row.getAttribute('data-wholesale-threshold')) || 0;
    const retailPrice = parseFloat(row.getAttribute('data-retail-price')) || 0;
    const wholesalePrice = parseFloat(row.getAttribute('data-wholesale-price')) || 0;
    const badgeSpan = row.querySelector('.price-badge');
    if (!isManualPriceChange) {
        if (wholesaleQty > 0 && qty >= wholesaleQty && wholesalePrice > 0) {
            currentPrice = wholesalePrice;
            badgeSpan.innerHTML = '<span class="bg-green-100 text-green-800 text-xs px-2 py-0.5 rounded font-semibold">Wholesale</span>';
        } else {
            currentPrice = retailPrice;
            badgeSpan.innerHTML = '<span class="bg-gray-200 text-gray-700 text-xs px-2 py-0.5 rounded">Retail</span>';
        }
        priceInput.value = currentPrice.toFixed(2);
    } else {
        badgeSpan.innerHTML = '<span class="bg-amber-100 text-amber-800 text-xs px-2 py-0.5 rounded font-semibold">Custom</span>';
    }
    const subtotal = qty * currentPrice;
    row.querySelector('.subtotal-td').innerText = 'KES ' + subtotal.toFixed(2);
    updateCartTotals();
}

function removeRow(btn) {
    btn.closest('tr').remove();
    updateCartTotals();
}

function openPaymentModal() {
    if (totalAmount <= 0) {
        alert('Cart is empty!');
        return;
    }
    document.getElementById('modal-total').innerText = 'KES ' + totalAmount.toFixed(2);
    document.getElementById('cash-input').value = totalAmount.toFixed(2);
    document.getElementById('mpesa-input').value = '0.00';
    calculateSplit();
    document.getElementById('payment-modal').classList.remove('hidden');
}

function closePaymentModal() {
    document.getElementById('payment-modal').classList.add('hidden');
}

function calculateSplit() {
    const cash = parseFloat(document.getElementById('cash-input').value) || 0;
    const mpesa = parseFloat(document.getElementById('mpesa-input').value) || 0;
    const sum = cash + mpesa;
    const statusBox = document.getElementById('payment-status');
    const confirmBtn = document.getElementById('confirm-sale-btn');
    if (sum < totalAmount - 0.009) {
        const remaining = totalAmount - sum;
        statusBox.className = "text-sm font-semibold text-center py-2 rounded-lg bg-amber-100 text-amber-800";
        statusBox.innerText = "Remaining: KES " + remaining.toFixed(2);
        confirmBtn.disabled = true;
    } else {
        const balanceAfterMpesa = Math.max(0, totalAmount - mpesa);
        const change = Math.max(0, cash - balanceAfterMpesa);
        if (change > 0) {
            statusBox.className = "text-sm font-semibold text-center py-2 rounded-lg bg-blue-100 text-blue-800";
            statusBox.innerText = "Change Due: KES " + change.toFixed(2);
        } else {
            statusBox.className = "text-sm font-semibold text-center py-2 rounded-lg bg-emerald-100 text-emerald-800";
            statusBox.innerText = "✓ Exact Amount Covered";
        }
        confirmBtn.disabled = false;
    }
}

async function submitSale() {
    const cash = parseFloat(document.getElementById('cash-input').value) || 0;
    const mpesa = parseFloat(document.getElementById('mpesa-input').value) || 0;
    const mpesaCode = document.getElementById('mpesa-code').value.trim();
    let paymentType = 'cash';
    if (cash > 0 && mpesa > 0) paymentType = 'split';
    else if (mpesa > 0) paymentType = 'mpesa';
    const rows = document.querySelectorAll('#cart-items tr');
    const items = [];
    rows.forEach(row => {
        const qtyInput = row.querySelector('.cart-qty-input');
        const priceInput = row.querySelector('.cart-price-input');
        const productId = parseInt(row.getAttribute('data-product-id')) || 0;
        if (qtyInput && priceInput && productId > 0) {
            const qty = parseInt(qtyInput.value) || 1;
            const unitPrice = parseFloat(priceInput.value) || 0;
            const subtotalTd = row.querySelector('.subtotal-td');
            const subtotal = subtotalTd ? parseFloat(subtotalTd.innerText.replace('KES', '').trim()) : (qty * unitPrice);
            items.push({ 
                product_id: productId, 
                quantity: qty, 
                unit_price: unitPrice, 
                subtotal: subtotal 
            });
        }
    });
    if (items.length === 0) {
        alert('No valid products found in cart.');
        return;
    }
    const payload = {
        total_amount: totalAmount,
        cash_amount: cash,
        mpesa_amount: mpesa,
        mpesa_code: mpesaCode,
        payment_type: paymentType,
        items: items
    };
    try {
        const response = await fetch('/api/sales/checkout', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(payload)
        });
        if (response.ok) {
            const result = await response.json();
            payload.change_given = result.change_given || 0;
            printThermalReceipt(payload, result.sale_id);
            document.getElementById('cart-items').innerHTML = '';
            updateCartTotals();
            closePaymentModal();
        } else {
            alert('Server error: ' + response.status);
        }
    } catch (err) {
        alert('Network error.');
        console.error(err);
    }
}

let cacheRecentSales = [];

async function openRecentSalesModal() {
    document.getElementById('recent-sales-modal').classList.remove('hidden');
    const container = document.getElementById('recent-sales-list');
    container.innerHTML = '<p class="text-gray-500 text-center py-6">Loading transactions...</p>';
    try {
        const res = await fetch('/api/sales/recent');
        if (!res.ok) throw new Error('Failed to load');
        cacheRecentSales = await res.json();
        if (!cacheRecentSales || cacheRecentSales.length === 0) {
            container.innerHTML = '<p class="text-gray-500 text-center py-6">No sales recorded yet.</p>';
            return;
        }
        let html = '';
        cacheRecentSales.forEach(function(sale) {
            const payBadge = sale.payment_type === 'mpesa' 
                ? '<span class="bg-green-100 text-green-800 text-xs px-2 py-0.5 rounded font-semibold">M-Pesa</span>'
                : (sale.payment_type === 'split' 
                    ? '<span class="bg-purple-100 text-purple-800 text-xs px-2 py-0.5 rounded font-semibold">Split</span>'
                    : '<span class="bg-blue-100 text-blue-800 text-xs px-2 py-0.5 rounded font-semibold">Cash</span>');
            html += '<div class="border border-gray-200 rounded-lg p-3 hover:bg-gray-50 transition flex justify-between items-center">' +
                '<div>' +
                    '<div class="flex items-center gap-2 mb-1">' +
                        '<span class="font-bold text-gray-900">Sale #' + sale.id + '</span>' +
                        payBadge +
                    '</div>' +
                    '<div class="text-xs text-gray-500">' + sale.created_at + ' • ' + (sale.items ? sale.items.length : 0) + ' item(s)</div>' +
                    '<div class="font-bold text-purple-900 text-sm mt-1">KES ' + sale.total_amount.toFixed(2) + '</div>' +
                '</div>' +
                '<button onclick="reprintFromLog(' + sale.id + ')" class="bg-gray-100 hover:bg-purple-600 hover:text-white text-gray-700 text-xs font-semibold px-3 py-2 rounded-lg border transition">🖨️ Reprint</button>' +
            '</div>';
        });
        container.innerHTML = html;
    } catch (err) {
        container.innerHTML = '<p class="text-red-500 text-center py-6">Error loading transaction history.</p>';
    }
}

function closeRecentSalesModal() {
    document.getElementById('recent-sales-modal').classList.add('hidden');
}

function reprintFromLog(saleId) {
    const sale = cacheRecentSales.find(s => s.id === saleId);
    if (!sale) return;
    const payload = {
        total_amount: sale.total_amount,
        cash_amount: sale.cash_amount,
        mpesa_amount: sale.mpesa_amount,
        mpesa_code: sale.mpesa_code,
        items: sale.items || []
    };
    printThermalReceipt(payload, sale.id, true);
}

function openAddProductModal() {
    const modal = document.getElementById('addProductModal');
    if (!modal) {
        alert('Add Product modal not found.');
        return;
    }
    modal.classList.remove('hidden');
    modal.classList.add('flex');
    loadCategories();
    document.getElementById('prodSku').focus();
}

function closeAddProductModal() {
    const modal = document.getElementById('addProductModal');
    if (modal) {
        modal.classList.add('hidden');
        modal.classList.remove('flex');
    }
    document.getElementById('addProductForm')?.reset();
    const alertBox = document.getElementById('modalAlert');
    if (alertBox) alertBox.classList.add('hidden');
}

async function loadCategories() {
    const select = document.getElementById('prodCategory');
    if (!select) return;
    select.innerHTML = '<option value="">Loading categories...</option>';
    try {
        const response = await fetch('/api/categories');
        const categories = await response.json();
        select.innerHTML = '<option value="">-- Select Category --</option>';
        if (Array.isArray(categories)) {
            categories.forEach(cat => {
                const opt = document.createElement('option');
                opt.value = cat.name;
                opt.textContent = cat.name;
                select.appendChild(opt);
            });
        }
        const addOpt = document.createElement('option');
        addOpt.value = 'NEW';
        addOpt.textContent = '➕ Add New Category...';
        addOpt.className = 'font-semibold text-purple-800 bg-purple-50';
        select.appendChild(addOpt);
    } catch (err) {
        select.innerHTML = '<option value="">Failed to load categories</option>';
    }
}

async function handleCategoryChange(selectElement) {
    if (selectElement.value === 'NEW') {
        const catName = prompt('Enter new category name:');
        if (catName && catName.trim() !== '') {
            try {
                const res = await fetch('/api/categories/create', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ name: catName.trim() })
                });
                if (res.ok) {
                    const newCat = await res.json();
                    await loadCategories();
                    selectElement.value = newCat.name;
                } else {
                    alert('Could not create category. It may already exist.');
                    selectElement.value = '';
                }
            } catch (e) {
                alert('Network error while adding category.');
                selectElement.value = '';
            }
        } else {
            selectElement.value = '';
        }
    }
}

async function saveProduct(event) {
    event.preventDefault();
    const saveBtn = document.getElementById('saveProductBtn');
    const alertBox = document.getElementById('modalAlert');
    const editId = document.getElementById('addProductForm')?.dataset.editId;
    const isEdit = editId && editId !== '';
    const wholesalePrice = parseFloat(document.getElementById('prodWholesalePrice')?.value) || 0;
    const wholesaleQty = parseInt(document.getElementById('prodWholesaleQty')?.value) || 0;
    if (wholesalePrice > 0 && wholesaleQty <= 0) {
        alertBox.className = 'p-3 rounded-lg text-sm font-medium bg-amber-100 text-amber-800 border border-amber-200';
        alertBox.innerText = 'Minimum wholesale quantity must be greater than 0 when wholesale price is set.';
        alertBox.classList.remove('hidden');
        return;
    }
    const payload = {
        barcode: document.getElementById('prodSku')?.value.trim(),
        name: document.getElementById('prodName')?.value.trim(),
        category: document.getElementById('prodCategory')?.value.trim(),
        cost_price: parseFloat(document.getElementById('prodBuyingPrice')?.value) || 0,
        retail_price: parseFloat(document.getElementById('prodSellingPrice')?.value) || 0,
        wholesale_price: wholesalePrice,
        wholesale_min_qty: wholesaleQty,
        stock_quantity: parseInt(document.getElementById('prodStockQty')?.value) || 0,
        reorder_level: parseInt(document.getElementById('prodReorderLevel')?.value) || 5
    };
    if (isEdit) {
        payload.id = parseInt(editId);
    }
    const url = isEdit ? '/api/products/update' : '/api/products/create';
    const method = isEdit ? 'PUT' : 'POST';
    saveBtn.disabled = true;
    saveBtn.innerText = 'Saving...';
    alertBox.classList.add('hidden');
    try {
        const response = await fetch(url, {
            method: method,
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(payload)
        });
        const result = await response.json();
        if (response.ok) {
            alertBox.className = 'p-3 rounded-lg text-sm font-medium bg-green-100 text-green-800 border border-green-200';
            alertBox.innerText = 'Product "' + result.name + '" ' + (isEdit ? 'updated' : 'created') + ' successfully!';
            alertBox.classList.remove('hidden');
            document.getElementById('addProductForm')?.reset();
            await loadProducts();
            setTimeout(() => closeAddProductModal(), 1500);
        } else {
            alertBox.className = 'p-3 rounded-lg text-sm font-medium bg-red-100 text-red-800 border border-red-200';
            alertBox.innerText = 'Error: ' + (result.error || 'Failed to ' + (isEdit ? 'update' : 'add') + ' product');
            alertBox.classList.remove('hidden');
        }
    } catch (error) {
        alertBox.className = 'p-3 rounded-lg text-sm font-medium bg-red-100 text-red-800 border border-red-200';
        alertBox.innerText = 'Network error. Is your Go backend running?';
        alertBox.classList.remove('hidden');
    } finally {
        saveBtn.disabled = false;
        saveBtn.innerText = 'Save Product';
        delete document.getElementById('addProductForm')?.dataset.editId;
        document.querySelector('#addProductModal h3').innerHTML = '📦 Add New Stock Item';
    }
}

async function loadProducts() {
    const tbody = document.getElementById('product-catalog-rows');
    if (!tbody) return;
    tbody.innerHTML = '<tr><td colspan="7" class="p-4 text-center text-gray-500">Loading products...</td></tr>';
    try {
        const response = await fetch('/api/products');
        if (!response.ok) throw new Error('Failed to load products');
        const products = await response.json();
        if (!products || products.length === 0) {
            tbody.innerHTML = '<tr><td colspan="7" class="p-4 text-center text-gray-500">No products found.</td></tr>';
            return;
        }
        let html = '';
        products.forEach(p => {
            const isLowStock = p.stock_quantity <= p.reorder_level;
            const stockBadge = isLowStock
                ? '<span class="bg-red-100 text-red-800 text-xs font-bold px-2 py-0.5 rounded-full">' + p.stock_quantity + ' (Low)</span>'
                : '<span class="bg-emerald-100 text-emerald-800 text-xs font-bold px-2 py-0.5 rounded-full">' + p.stock_quantity + '</span>';
            html += '<tr class="border-b hover:bg-purple-50/50 transition">' +
                '<td class="p-3 font-mono text-xs text-gray-500 font-semibold">' + (p.barcode || 'N/A') + '</td>' +
                '<td class="p-3 font-semibold text-gray-900">' + p.name + '</td>' +
                '<td class="p-3 text-gray-500 text-xs">' + (p.category || 'General') + '</td>' +
                '<td class="p-3 text-right font-mono font-bold text-purple-900">KES ' + (p.retail_price ? p.retail_price.toFixed(2) : '0.00') + '</td>' +
                '<td class="p-3 text-right font-mono text-gray-700">KES ' + (p.wholesale_price ? p.wholesale_price.toFixed(2) : '0.00') + '</td>' +
                '<td class="p-3 text-center">' + stockBadge + '</td>' +
                '<td class="p-3 text-center"><button onclick=\'editProduct(' + JSON.stringify(p) + ')\' class="bg-blue-100 hover:bg-blue-200 text-blue-800 text-xs font-bold px-2.5 py-1 rounded-lg transition">✏️ Edit</button></td>' +
            '</tr>';
        });
        tbody.innerHTML = html;
    } catch (err) {
        tbody.innerHTML = '<tr><td colspan="7" class="p-4 text-center text-red-600 font-mono text-xs">⚠️ ' + err.message + '</td></tr>';
    }
}

async function saveExpense(event) {
    event.preventDefault();
    const saveBtn = document.getElementById('saveExpenseBtn');
    const alertBox = document.getElementById('expenseAlert');
    const payload = {
        category: document.getElementById('expenseCategory')?.value,
        description: document.getElementById('expenseDescription')?.value.trim(),
        amount: parseFloat(document.getElementById('expenseAmount')?.value) || 0,
        expense_date: document.getElementById('expenseDate')?.value,
        payment_method: document.getElementById('expensePaymentMethod')?.value || 'cash',
        created_by: 'Cashier'
    };
    if (!payload.category || payload.amount <= 0) {
        alertBox.className = 'p-3 rounded-lg text-sm font-medium bg-amber-100 text-amber-800 border border-amber-200';
        alertBox.innerText = payload.category ? 'Amount must be greater than 0.' : 'Please select a category.';
        alertBox.classList.remove('hidden');
        return;
    }
    saveBtn.disabled = true;
    saveBtn.innerText = 'Saving...';
    alertBox.classList.add('hidden');
    try {
        const response = await fetch('/api/expenses/create', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(payload)
        });
        const result = await response.json();
        if (response.ok) {
            alertBox.className = 'p-3 rounded-lg text-sm font-medium bg-green-100 text-green-800 border border-green-200';
            alertBox.innerText = 'Expense recorded successfully!';
            alertBox.classList.remove('hidden');
            document.getElementById('expenseForm')?.reset();
            setTimeout(() => closeExpenseModal(), 1500);
        } else {
            alertBox.className = 'p-3 rounded-lg text-sm font-medium bg-red-100 text-red-800 border border-red-200';
            alertBox.innerText = 'Error: ' + (result.error || 'Failed to save expense');
            alertBox.classList.remove('hidden');
        }
    } catch (error) {
        alertBox.className = 'p-3 rounded-lg text-sm font-medium bg-red-100 text-red-800 border border-red-200';
        alertBox.innerText = 'Network error. Please try again.';
        alertBox.classList.remove('hidden');
    } finally {
        saveBtn.disabled = false;
        saveBtn.innerText = 'Save Expense';
    }
}

async function fetchInventoryValuation() {
    const tbody = document.getElementById('valuation-rows');
    if (!tbody) return;
    tbody.innerHTML = '<tr><td colspan="7" class="p-4 text-center text-gray-500">Calculating inventory metrics...</td></tr>';
    closeDrilldown();

    try {
        const response = await fetch('/api/reports/inventory-valuation');
        if (!response.ok) {
            const errText = await response.text();
            throw new Error('Server Error (' + response.status + '): ' + errText);
        }

        const items = await response.json();
        if (!items || items.length === 0) {
            tbody.innerHTML = '<tr><td colspan="7" class="p-4 text-center text-gray-500">No inventory products found.</td></tr>';
            return;
        }

        let sumCost = 0, sumRetail = 0, sumProfit = 0;
        items.forEach(item => {
            sumCost += item.total_cost;
            sumRetail += item.total_retail;
            sumProfit += item.potential_profit;
        });

        let html = '';
        items.forEach(item => {
            const capitalSharePct = sumCost > 0 ? (item.total_cost / sumCost) * 100 : 0;
            const marginPct = item.total_retail > 0 ? (item.potential_profit / item.total_retail) * 100 : 0;

            html += '<tr class="border-b hover:bg-indigo-50/50 transition">' +
                '<td class="p-3 font-semibold text-gray-900">' + item.category + '</td>' +
                '<td class="p-3 text-center"><div class="flex items-center gap-2"><div class="w-full bg-gray-200 rounded-full h-2 overflow-hidden"><div class="bg-indigo-600 h-2 rounded-full" style="width: ' + capitalSharePct.toFixed(1) + '%"></div></div><span class="text-[11px] font-mono font-bold text-gray-600 w-12 text-right">' + capitalSharePct.toFixed(1) + '%</span></div></td>' +
                '<td class="p-3 text-center font-bold text-gray-800">' + item.total_quantity + '</td>' +
                '<td class="p-3 text-right font-mono text-gray-700">KES ' + item.total_cost.toFixed(2) + '</td>' +
                '<td class="p-3 text-right font-mono font-semibold text-purple-900">KES ' + item.total_retail.toFixed(2) + '</td>' +
                '<td class="p-3 text-right font-mono font-bold text-emerald-700">' + marginPct.toFixed(1) + '%</td>' +
                '<td class="p-3 text-center"><button onclick="loadCategoryDrilldown(\'' + encodeURIComponent(item.category) + '\')" class="bg-indigo-100 hover:bg-indigo-200 text-indigo-800 font-bold text-xs px-2.5 py-1 rounded-lg transition cursor-pointer">🔍 View Items</button></td>' +
            '</tr>';
        });

        tbody.innerHTML = html;

        const overallMarginPct = sumRetail > 0 ? (sumProfit / sumRetail) * 100 : 0;
        document.getElementById('val-total-cost').innerText = 'KES ' + sumCost.toFixed(2);
        document.getElementById('val-total-retail').innerText = 'KES ' + sumRetail.toFixed(2);
        document.getElementById('val-total-profit').innerText = 'KES ' + sumProfit.toFixed(2);
        document.getElementById('val-avg-margin').innerText = overallMarginPct.toFixed(1) + '%';

    } catch (err) {
        tbody.innerHTML = '<tr><td colspan="7" class="p-4 text-center text-red-600 font-mono text-xs">⚠️ ' + err.message + '</td></tr>';
    }
}

async function loadCategoryDrilldown(encodedCategory) {
    const categoryName = decodeURIComponent(encodedCategory);
    const panel = document.getElementById('valuation-drilldown-panel');
    const tbody = document.getElementById('drilldown-rows');
    const title = document.getElementById('drilldown-category-title');

    if (!panel || !tbody || !title) return;

    title.innerText = 'Product Breakdown: ' + categoryName;
    tbody.innerHTML = '<tr><td colspan="5" class="p-3 text-center text-gray-500">Loading products...</td></tr>';
    panel.classList.remove('hidden');

    try {
        const res = await fetch('/api/reports/inventory-valuation/details?category=' + encodeURIComponent(categoryName));
        if (!res.ok) throw new Error(await res.text());

        const products = await res.json();
        if (!products || products.length === 0) {
            tbody.innerHTML = '<tr><td colspan="5" class="p-3 text-center text-gray-500">No products found for this category.</td></tr>';
            return;
        }

        let html = '';
        products.forEach(p => {
            html += '<tr class="border-b hover:bg-gray-50">' +
                '<td class="p-2 font-semibold text-gray-800">' + p.product_name + '</td>' +
                '<td class="p-2 text-center font-bold text-gray-900">' + p.in_stock + '</td>' +
                '<td class="p-2 text-right font-mono">KES ' + p.cost_price.toFixed(2) + '</td>' +
                '<td class="p-2 text-right font-mono">KES ' + p.retail_price.toFixed(2) + '</td>' +
                '<td class="p-2 text-right font-mono font-bold text-indigo-900">KES ' + p.total_cost.toFixed(2) + '</td>' +
            '</tr>';
        });

        tbody.innerHTML = html;
    } catch (err) {
        tbody.innerHTML = '<tr><td colspan="5" class="p-3 text-center text-red-600 font-mono text-xs">⚠️ ' + err.message + '</td></tr>';
    }
}
